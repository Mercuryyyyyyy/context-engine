## IndexedDB 缓存封装

```typescript
const DB_NAME = 'app-cache';
const STORE = 'kv';
const TTL_MS = 30 * 60 * 1000;

interface CacheEntry<T> {
  key: string;
  data: T;
  expiresAt: number;
}

// 打开数据库（单例）
let dbPromise: Promise<IDBDatabase> | null = null;
function openDB(): Promise<IDBDatabase> {
  if (dbPromise) return dbPromise;
  dbPromise = new Promise((resolve, reject) => {
    const req = indexedDB.open(DB_NAME, 1);
    req.onupgradeneeded = () => {
      req.result.createObjectStore(STORE, { keyPath: 'key' });
    };
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => reject(req.error);
  });
  return dbPromise;
}

// 通用事务助手
async function tx<T>(
  mode: IDBTransactionMode,
  fn: (store: IDBObjectStore) => IDBRequest<T>
): Promise<T> {
  const db = await openDB();
  return new Promise((resolve, reject) => {
    const t = db.transaction(STORE, mode);
    const req = fn(t.objectStore(STORE));
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => reject(req.error);
  });
}

// 批量写入
async function setMany<T>(entries: { key: string; data: T }[]): Promise<void> {
  const db = await openDB();
  return new Promise((resolve, reject) => {
    const t = db.transaction(STORE, 'readwrite');
    const store = t.objectStore(STORE);
    const expiresAt = Date.now() + TTL_MS;
    for (const { key, data } of entries) {
      store.put({ key, data, expiresAt } satisfies CacheEntry<T>);
    }
    t.oncomplete = () => resolve();
    t.onerror = () => reject(t.error);
  });
}

// 批量读取（自动过滤过期）
async function getMany<T>(keys: string[]): Promise<Map<string, T>> {
  const db = await openDB();
  return new Promise((resolve, reject) => {
    const t = db.transaction(STORE, 'readonly');
    const store = t.objectStore(STORE);
    const result = new Map<string, T>();
    const now = Date.now();
    for (const key of keys) {
      const req = store.get(key);
      req.onsuccess = () => {
        const entry = req.result as CacheEntry<T> | undefined;
        if (entry && entry.expiresAt > now) result.set(key, entry.data);
      };
    }
    t.oncomplete = () => resolve(result);
    t.onerror = () => reject(t.error);
  });
}

// 带缓存的 fetch
async function getUsersCached(): Promise<User[]> {
  const key = 'users';
  const cached = await getMany<User[]>([key]);
  if (cached.has(key)) return cached.get(key)!;

  const users = await fetchWithRetry('/api/users');
  await setMany([{ key, data: users }]);
  return users;
}
```

## 要点

- **单例 `dbPromise`**：避免重复打开数据库。
- **`keyPath: 'key'`**：天然支持批量 `put`/`get`。
- **过期过滤在读取时**：惰性清理，无需定时任务；也可加 `onupgradeneeded` 或后台清理。
- **`satisfies`**：保留字面量类型同时校验结构。
- **事务完成用 `oncomplete`**：比逐个 `onsuccess` 更可靠，确保写入落盘。

## IndexedDB 缓存封装

```typescript
const DB_NAME = 'app-cache';
const STORE = 'kv';
const TTL_MS = 30 * 60 * 1000;

interface CacheEntry<T> {
  key: string;
  data: T;
  expiresAt: number;
}

// 打开数据库（单例）
let dbPromise: Promise<IDBDatabase> | null = null;
function openDB(): Promise<IDBDatabase> {
  if (dbPromise) return dbPromise;
  dbPromise = new Promise((resolve, reject) => {
    const req = indexedDB.open(DB_NAME, 1);
    req.onupgradeneeded = () => {
      const db = req.result;
      if (!db.objectStoreNames.contains(STORE)) {
        db.createObjectStore(STORE, { keyPath: 'key' });
      }
    };
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => reject(req.error);
  });
  return dbPromise;
}

// 通用事务助手
async function withStore<T>(
  mode: IDBTransactionMode,
  fn: (store: IDBObjectStore) => IDBRequest<T>
): Promise<T> {
  const db = await openDB();
  return new Promise((resolve, reject) => {
    const tx = db.transaction(STORE, mode);
    const req = fn(tx.objectStore(STORE));
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => reject(req.error);
  });
}

// 批量写入
async function setMany<T>(entries: Array<{ key: string; data: T }>): Promise<void> {
  const db = await openDB();
  const expiresAt = Date.now() + TTL_MS;
  await new Promise<void>((resolve, reject) => {
    const tx = db.transaction(STORE, 'readwrite');
    const store = tx.objectStore(STORE);
    for (const { key, data } of entries) {
      store.put({ key, data, expiresAt } satisfies CacheEntry<T>);
    }
    tx.oncomplete = () => resolve();
    tx.onerror = () => reject(tx.error);
  });
}

// 批量读取（自动过滤过期项）
async function getMany<T>(keys: string[]): Promise<Map<string, T>> {
  const db = await openDB();
  const now = Date.now();
  const result = new Map<string, T>();
  const expired: string[] = [];

  await new Promise<void>((resolve, reject) => {
    const tx = db.transaction(STORE, 'readwrite');
    const store = tx.objectStore(STORE);
    for (const key of keys) {
      const req = store.get(key);
      req.onsuccess = () => {
        const entry = req.result as CacheEntry<T> | undefined;
        if (!entry) return;
        if (entry.expiresAt < now) expired.push(key);
        else result.set(key, entry.data);
      };
    }
    tx.oncomplete = () => {
      for (const key of expired) store.delete(key); // 惰性清理
      resolve();
    };
    tx.onerror = () => reject(tx.error);
  });

  return result;
}

## 观察者模式缓存层

```typescript
// 1. 通用 EventEmitter
type Listener<T> = (payload: T) => void;

class EventEmitter<Events extends Record<string, unknown>> {
  private listeners = new Map<keyof Events, Set<Listener<any>>>();

  on<K extends keyof Events>(event: K, fn: Listener<Events[K]>): () => void {
    let set = this.listeners.get(event);
    if (!set) this.listeners.set(event, (set = new Set()));
    set.add(fn);
    return () => set!.delete(fn); // 返回取消订阅函数
  }

  protected emit<K extends keyof Events>(event: K, payload: Events[K]): void {
    this.listeners.get(event)?.forEach(fn => fn(payload));
  }
}

// 2. CacheStore
interface CacheEvents {
  change: { key: string; data: unknown };
  invalidate: { key: string };
}

class CacheStore extends EventEmitter<CacheEvents> {
  private memory = new Map<string, unknown>();

  async set<T>(key: string, data: T): Promise<void> {
    this.memory.set(key, data);
    await setMany([{ key, data }]);
    this.emit('change', { key, data });
  }

  async get<T>(key: string): Promise<T | undefined> {
    if (this.memory.has(key)) return this.memory.get(key) as T;
    const map = await getMany<T>([key]);
    const data = map.get(key);
    if (data !== undefined) this.memory.set(key, data);
    return data;
  }

  async invalidate(key: string): Promise<void> {
    this.memory.delete(key);
    await tx('readwrite', s => s.delete(key));
    this.emit('invalidate', { key });
  }
}

export const cacheStore = new CacheStore();

## 轻量 CLI 解析器设计

核心思路：**声明式定义 → 解析 → 类型推导**。用泛型从 schema 推导出参数类型。

```typescript
// 1. 选项定义
type OptionType = 'string' | 'number' | 'boolean';

interface OptionDef<T extends OptionType = OptionType> {
  type: T;
  alias?: string;
  default?: unknown;
  desc?: string;
}

type InferOption<D extends OptionDef> =
  D['type'] extends 'number' ? number :
  D['type'] extends 'boolean' ? boolean : string;

// 从 schema 推导整个 options 对象类型
type InferOptions<S extends Record<string, OptionDef>> = {
  [K in keyof S]: InferOption<S[K]>;
};

// 2. 命令定义
interface CommandDef<S extends Record<string, OptionDef>> {
  options: S;
  run: (opts: InferOptions<S>, positional: string[]) => void | Promise<void>;
}

// 3. 解析器
function parseArgs<S extends Record<string, OptionDef>>(
  argv: string[],
  schema: S
): { options: InferOptions<S>; positional: string[] } {
  const options = {} as Record<string, unknown>;
  const positional: string[] = [];

  // 建立 alias → 主名 映射
  const aliasMap = new Map<string, string>();
  for (const [name, def] of Object.entries(schema)) {
    if (def.default !== undefined) options[name] = def.default;
    if (def.alias) aliasMap.set(def.alias, name);
  }

  for (let i = 0; i < argv.length; i++) {
    const arg = argv[i];
    if (!arg.startsWith('-')) { positional.push(arg); continue; }

    const isLong = arg.startsWith('--');
    const raw = isLong ? arg.slice(2) : arg.slice(1);
    const [flag, inlineVal] = raw.split('=');
    const name = isLong ? flag : (aliasMap.get(flag) ?? flag);
    const def = schema[name];
    if (!def) throw new Error(`Unknown option: ${arg}`);

    if (def.type === 'boolean') {
      options[name] = inlineVal !== undefined ? inlineVal !== 'false' : true;
    } else {
      const value = inlineVal ?? argv[++i];
      if (value === undefined) throw new Error(`Missing value for ${arg}`);
      options[name] = def.type === 'number' ? Number(value) : value;
    }
  }
  return { options: options as InferOptions<S>, positional };
}

const KEY = {
  up: '\u001B[A', down: '\u001B[B',
  right: '\u001B[C', left: '\u001B[D',
  enter: '\r', space: ' ', esc: '\u001B',

function readKeys(onKey: (key: string) => void): () => void {

    stdin.setRawMode(wasRaw ?? false);
    stdin.pause();
    stdin.off('data', handler);
  };
  stdin.on('data', handler);
  return cleanup;
}

async function select(message: string, choices: string[]): Promise<string> {
  let index = 0;
  stdin.setRawMode(true);
  stdin.resume();

  const draw = () => {
    stdout.write('\x1b[?25l');
    const lines = [message, ...choices.map((c, i) =>
      i === index ? `\x1b[36m❯ ${c}\x1b[0m` : `  ${c}`)];
    stdout.write('\x1b[2K\x1b[G' + lines.join('\n'));
  };

  draw();
  while (true) {
    const key = await readKey();
    if (key === KEYS.ctrlC) { cleanup(); process.exit(0); }
    if (key === KEYS.up) index = (index - 1 + choices.length) % choices.length;
    else if (key === KEYS.down) index = (index + 1) % choices.length;
    else if (key === KEYS.enter) break;
    // 回到顶部重绘
    stdout.write(`\x1b[${choices.length + 1}A`);
    draw();
  }
  cleanup();
  return choices[index];
}

function cleanup() {
  stdin.setRawMode(false);
  stdin.pause();
  stdout.write('\x1b[?25h\n'); // 恢复光标
}

async function multiSelect(message: string, choices: string[]): Promise<string[]> {
  let index = 0;
  const selected = new Set<number>();
  stdin.setRawMode(true); stdin.resume();

  const draw = () => {
    const lines = [message, ...choices.map((c, i) => {
      const cursor = i === index ? '\x1b[36m❯\x1b[0m' : ' ';
      const box = selected.has(i) ? '\x1b[32m◉\x1b[0m' : '◯';
      return `${cursor} ${box} ${c}`;
    })];
    stdout.write('\x1b[2K\x1b[G' + lines.join('\n'));
  };

  draw();
  while (true) {
    const key = await readKey();
    if (key === KEYS.ctrlC) { cleanup(); process.exit(0); }
    if (key === KEYS.up) index = (index - 1 + choices.length) % choices.length;
    else if (key === KEYS.down) index = (index + 1) % choices.length;
    else if (key === KEYS.space) selected.has(index) ? selected.delete(index) : selected.add(index);
    else if (key === KEYS.enter) break;
    stdout.write(`\x1b[${choices.length + 1}A`);
    draw();
  }
  cleanup();
  return [...selected].map(i => choices[i]);
}

## 关键点

- **raw mode**：`setRawMode(true)` 关闭行缓冲，逐键读取；退出时务必恢复，否则终端会卡住。
- **ANSI 转义**：`\x1b[A/B` 上下移动，`\x1b[2K\x1b[G` 清行回列首，`\x1b[?25l/h` 隐藏/显示光标。
- **重绘技巧**：用 `\x1b[NA` 把光标移回菜单顶部再重画，避免闪烁。
- **Ctrl+C**：raw mode 下不再自动退出，需手动捕获 `\x03`。
- **Backspace**：`\x7f`，用 `\b \b` 擦除。

## 注意事项

- 多字节输入（中文、粘贴）会一次返回多个字符，生产级需处理 `readline` 的 `keypress` 事件或 `string-width`。
- 终端宽度、换行、颜色兼容性（Windows 老终端）需额外处理。
- 实际项目建议用 **`@inquirer/prompts`** 或 **`prompts`**，它们已处理上述边界情况。

## 进阶优化方案

### 1. 定位真实瓶颈

先别急着改，用 **React DevTools Profiler** 录制滚动，看：
- 是 **render 耗时** 还是 **commit/paint 耗时**？
- 哪些组件在 re-render？

FPS 20 通常是**主线程被 render 或图片解码占满**。

### 2. 行组件拆分 + 隔离

复杂行拆成独立子组件，各自 `memo`，避免整行重渲染：

```tsx
const Row = memo(({ item }: { item: Item }) => (
  <div className="row">
    <RowHeader item={item} />   {/* memo */}
    <RowChart data={item.chart} /> {/* 懒加载 + memo */}
    <RowActions id={item.id} />  {/* memo，回调用 useCallback */}
  </div>
));
```

**关键**：传给子组件的 props 必须稳定。回调用 `useCallback`，对象用 `useMemo`，否则 `memo` 失效。

### 3. 图表是重灾区

每行一个 echarts 实例，10000 行滚动时频繁创建/销毁，极耗性能。

**方案 A：图表延迟渲染**——只在行**停止滚动后**渲染：

```tsx
function useIsScrolling(delay = 150) {
  const [scrolling, setScrolling] = useState(false);
  const timer = useRef<number>();
  useEffect(() => {
    const onScroll = () => {
      setScrolling(true);
      clearTimeout(timer.current);
      timer.current = window.setTimeout(() => setScrolling(false), delay);
    };
    window.addEventListener('scroll', onScroll, { passive: true });
    return () => window.removeEventListener('scroll', onScroll);
  }, [delay]);
  return scrolling;
}

// 滚动中显示占位，停止后渲染图表
{scrolling ? <ChartPlaceholder /> : <RowChart data={item.chart} />}
```

**方案 B：用轻量图表**——echarts 太重，改用 **轻量 canvas 绘制**或 **SVG sparkline**，甚至纯 CSS 柱状图。

**方案 C：图表实例复用**——用 canvas 池，滚动时复用同一批 canvas 重绘。

### 4. 图片解码优化

高清图解码是主线程杀手。

```tsx
<img
  src={thumbUrl}          // 列表用缩略图，别用原图
  loading="lazy"
  decoding="async"        // 异步解码，不阻塞主线程
  width={80} height={80}  // 固定尺寸，避免布局抖动
/>
```

- **缩略图**：列表用小图（如 80×80），点击详情再加载原图。
- **`decoding="async"`**：让浏览器在后台线程解码。
- **`content-visibility: auto`**：让浏览器跳过屏幕外元素的渲染：

```css
.row { content-visibility: auto; contain-intrinsic-size: 0 80px; }
```

### 5. 减少 re-render

**用 `useSyncExternalStore` 或 selector 订阅**，避免父组件状态变化导致全列表重渲染：

```tsx
// 若用 Zustand
const selectedId = useStore(s => s.selectedId); // 只订阅需要的字段
```

**react-window 的 `itemData`**：传稳定引用，避免每次 render 新建对象。

### 6. 滚动性能细节

- **`passive: true`**：滚动监听加 passive，避免阻塞。
- **避免滚动中 setState**：用 `requestAnimationFrame` 节流。
- **`will-change: transform`**：谨慎使用，过多反而拖慢。
- **CSS `contain: strict`**：限制行的重排范围。

### 7. 终极方案：Canvas 渲染

若行内容高度同质（如都是图表），考虑**整列表用 canvas 绘制**（类似 Google Sheets），完全绕过 DOM。代价是失去 DOM 交互，需自行处理点击命中。

| 优先级 | 措施 | 预期收益 |
|--------|------|---------|
| P0 | 图表延迟渲染（滚动中占位） | 大幅减少 render |
| P0 | 图片用缩略图 + `decoding="async"` | 减少解码阻塞 |
| P1 | 行拆分子组件 + memo + 稳定 props | 减少 re-render |
| P1 | `content-visibility: auto` | 跳过屏外渲染 |

## 进阶优化方案

FPS 20 说明主线程被**渲染 + 图片解码**占满。按收益排序：

### 1. 行组件拆分 + 内容分级渲染

复杂行是主因。把行拆成**骨架层**和**内容层**，滚动时只渲染骨架：

```tsx
const Row = React.memo(({ index, style, data }) => {
  const item = data[index];
  const [ready, setReady] = useState(false);

  // 滚动停止后再渲染重内容
  useEffect(() => {
    const id = setTimeout(() => setReady(true), 100);
    return () => clearTimeout(id);
  }, []);

  return (
    <div style={style}>
      {ready ? <RowContent item={item} /> : <RowSkeleton />}
    </div>
  );
});
```

配合 `useDeferredValue` 或滚动状态，滚动中降级、停止后升级。

### 2. 图表延迟 + 复用

每行一个 echarts 实例是灾难。方案：

- **滚动中不渲染图表**，只显示占位。
- **图表实例池**：只保留可视区附近的实例，滚出即销毁。
- 或用**轻量 SVG 迷你图**替代 echarts（如手写 sparkline）。

```tsx
const MiniChart = React.memo(({ data }) => {
  // 纯 SVG，无 canvas 初始化开销
  const path = useMemo(() => buildPath(data), [data]);
  return <svg viewBox="0 0 100 30"><path d={path} /></svg>;
});
```

### 3. 图片优化（解码是隐藏杀手）

高清图解码在主线程，是掉帧主因：

```tsx

  src={thumbUrl}          // 用缩略图，别用原图

- 考虑 **WebP/AVIF**，解码更快、体积更小。

### 4. 减少 re-render

`React.memo` 只挡 props 变化，但 `data` 数组引用变化会让所有行失效：

```tsx

// ✅ 稳定引用

```

const handleClick = useCallback((id) => {...}, []);

const onScroll = useMemo(

  []

### 6. CSS 优化

- 行内避免 `box-shadow`、`filter`、`backdrop-filter`（触发重绘）。
- 用 `transform` / `opacity` 做动画（走合成层）。
- 给行加 `will-change: transform`（谨慎，过多反而慢）。
- `contain: layout paint` 隔离每行渲染：

```css

- **@tanstack/react-virtual**：更灵活，支持动态测量。
- **react-virtuoso**：内置图片加载、动态高度处理。
